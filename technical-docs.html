Database Schema

-- Transactions table
CREATE TABLE transactions (
    id SERIAL PRIMARY KEY,
    date_mutation DATE,
    valeur_fonciere DECIMAL(15,2),
    code_postal VARCHAR(10),
    code_commune VARCHAR(10),
    code_departement VARCHAR(5),
    type_local VARCHAR(50),
    surface_reelle_bati DECIMAL(10,2),
    price_per_m2 DECIMAL(10,2),
    geom GEOMETRY(Point, 4326)
);

-- Spatial index for fast queries
CREATE INDEX idx_transactions_geom ON transactions USING GIST(geom);

-- Aggregates table
CREATE TABLE price_aggregates (
    id SERIAL PRIMARY KEY,
    level VARCHAR(20),
    code VARCHAR(50),
    name VARCHAR(255),
    median_price DECIMAL(10,2),
    transaction_count INTEGER,
    confidence_score INTEGER,
    geom GEOMETRY(MULTIPOLYGON, 4326)
);

Cadastre Building Matching

For each transaction point, find the containing building polygon from cadastre data:

# Build spatial index on cadastre
from shapely import STRtree
tree = STRtree(buildings_gdf.geometry)

# Query for containing building
candidates = tree.query(transaction_point)
for idx in candidates:
    if buildings_gdf.geometry.iloc[idx].contains(transaction_point):
        return buildings_gdf.geometry.iloc[idx]

Hierarchical Smoothing Formula

# Weight based on transaction count
weight = min(1.0, n / 50)

# Smoothed estimate
smoothed = weight × local_median + (1 - weight) × parent_median

Confidence Score Calculation

def calculate_confidence(n, ci_width, median, recency_months):
    # Volume component (0-40 points)
    volume_score = min(40, n * 2)
    
    # Precision component (0-30 points)
    relative_ci = ci_width / median
    precision_score = max(0, 30 - relative_ci * 50)
    
    # Recency component (0-30 points)
    recency_score = max(0, 30 - recency_months)
    
    return volume_score + precision_score + recency_score
Confidence Score Range Meaning
High 70-100 Reliable estimate
Medium 50-69 Moderate uncertainty
Low 30-49 High uncertainty
Estimated 0-29 Heavily smoothed

Spatial Join SQL

SELECT 
    zones.code,
    zones.name,
    COUNT(*) as transaction_count,
    PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY t.price_per_m2) as median
FROM zones
JOIN transactions t ON ST_Within(t.geom, zones.geom)
GROUP BY zones.code, zones.name;
